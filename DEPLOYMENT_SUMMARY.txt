   cd /Users/diosplan/Documents/sessions-smart-contract
   scarb build

2. Declare the production contract:
   starkli declare target/dev/sessions_smart_contract_Account.contract_class.json \
     --account ~/.starkli-wallets/deployer/account.json \
     --keystore ~/.starkli-wallets/deployer/keystore.json \
     --rpc https://starknet-mainnet.public.blastapi.io/rpc/v0_7

3. Deploy an instance:
   starkli deploy <CLASS_HASH> <YOUR_OWNER_PUBLIC_KEY> \
     --account ~/.starkli-wallets/deployer/account.json \
     --keystore ~/.starkli-wallets/deployer/keystore.json \
     --rpc https://starknet-mainnet.public.blastapi.io/rpc/v0_7

4. Verify SNIP-9 compatibility:
   starkli call <CONTRACT_ADDRESS> get_snip9_version
   # Should return: 2

5. Test with Paymaster:
   # See SNIP9_PAYMASTER_GUIDE.md for examples

6. Update README.md with new class hashes
